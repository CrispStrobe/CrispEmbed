cmake_minimum_required(VERSION 3.16)
project(lighton4b_swiglu_source_probe LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED GGML_SOURCE)
 message(FATAL_ERROR "Exact GGML_SOURCE required")
endif()
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(swiglu-source-probe main.cpp)
target_link_libraries(swiglu-source-probe PRIVATE ggml)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(swiglu-source-probe PRIVATE -ffp-contract=off)
endif()

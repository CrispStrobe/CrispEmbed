cmake_minimum_required(VERSION 3.16)
project(lighton4b_physical_primitive LANGUAGES C CXX)
if(NOT DEFINED GGML_SOURCE OR NOT EXISTS "${GGML_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "An authenticated pinned ggml source is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
# Match both actual parent builds; standalone ggml otherwise defaults OFF.
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(physical-primitive main.cpp)
target_link_libraries(physical-primitive PRIVATE ggml ggml-cpu)

cmake_minimum_required(VERSION 3.16)
project(lighton4b_attention_primitive LANGUAGES C CXX)
if(NOT DEFINED GGML_SOURCE OR NOT EXISTS "${GGML_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "An authenticated pinned ggml source is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
# Match both actual parent builds; standalone ggml otherwise defaults OFF.
option(ATTENTION_DIAGNOSTIC_LLAMAFILE "Diagnostic dispatch control; actual profile is ON" ON)
set(GGML_LLAMAFILE ${ATTENTION_DIAGNOSTIC_LLAMAFILE} CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(attention-primitive graph.cpp)
target_link_libraries(attention-primitive PRIVATE ggml ggml-cpu)

# Private backend symbol experiment: explicit original F16 dot, no trait mutation.
add_executable(attention-paired-f16 paired.cpp)
target_link_libraries(attention-paired-f16 PRIVATE ggml ggml-cpu)

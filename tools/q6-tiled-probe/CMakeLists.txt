cmake_minimum_required(VERSION 3.16)
project(lighton4b_q6_tiled LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
if(NOT DEFINED GGML_SOURCE OR NOT DEFINED PROBE_OPENMP OR NOT DEFINED PROBE_LLAMAFILE)
 message(FATAL_ERROR "Pinned GGML source and explicit actual OpenMP profile required")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_FMA ON CACHE BOOL "" FORCE)
set(GGML_F16C ON CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ${PROBE_LLAMAFILE} CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ${PROBE_OPENMP} CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(lighton4b-q6-tiled-probe probe.cpp ../../src/lighton4b_physical_q6_tiled.cpp)
target_include_directories(lighton4b-q6-tiled-probe PRIVATE ../../src)
target_compile_options(lighton4b-q6-tiled-probe PRIVATE -ffp-contract=off)
target_link_libraries(lighton4b-q6-tiled-probe PRIVATE ggml ggml-base ggml-cpu Threads::Threads)

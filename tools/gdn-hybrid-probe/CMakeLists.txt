cmake_minimum_required(VERSION 3.16)
project(lighton4b_gdn_trace_replay LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED GGML_SOURCE)
 message(FATAL_ERROR "GGML_SOURCE must be the authenticated pinned implementation")
endif()
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_executable(gdn-hybrid-probe main.cpp)
target_link_libraries(gdn-hybrid-probe PRIVATE ggml)
if(DEFINED NATIVE_SOURCE)
 target_sources(gdn-hybrid-probe PRIVATE "${NATIVE_SOURCE}/src/qwen35_delta.cpp" "${NATIVE_SOURCE}/src/lighton4b_physical_gdn_source_ops.cpp")
 target_include_directories(gdn-hybrid-probe PRIVATE "${NATIVE_SOURCE}/src")
 target_compile_definitions(gdn-hybrid-probe PRIVATE CRISPEMBED_GDN_RUNTIME_PROBE=1)
endif()
# The actual native helper is compiled with contraction disabled in root CMake.
# Keep this standalone probe and its scalar reconstruction on that exact rule.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(gdn-hybrid-probe PRIVATE -ffp-contract=off)
endif()

if(DEFINED NATIVE_SOURCE)
 add_executable(gdn-hybrid-cache-controls "${NATIVE_SOURCE}/tests/test_lighton4b_physical_gdn_source_ops.cpp" "${NATIVE_SOURCE}/src/qwen35_delta.cpp" "${NATIVE_SOURCE}/src/lighton4b_physical_gdn_source_ops.cpp" "${NATIVE_SOURCE}/src/qwen35_linear.cpp" "${NATIVE_SOURCE}/src/qwen35_norm.cpp" "${NATIVE_SOURCE}/src/qwen35_profile.cpp" "${NATIVE_SOURCE}/src/lighton4b_physical_f16_attention.cpp")
 target_include_directories(gdn-hybrid-cache-controls PRIVATE "${NATIVE_SOURCE}/src")
 target_link_libraries(gdn-hybrid-cache-controls PRIVATE ggml)
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gdn-hybrid-cache-controls PRIVATE -ffp-contract=off)
 endif()
endif()

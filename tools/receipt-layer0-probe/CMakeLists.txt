cmake_minimum_required(VERSION 3.16)
project(lighton4b_receipt_layer0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED CRISPEMBED_SOURCE)
 message(FATAL_ERROR "Exact candidate source required")
endif()
add_subdirectory("${CRISPEMBED_SOURCE}" product EXCLUDE_FROM_ALL)
add_executable(receipt-layer0-component component.cpp)
target_compile_options(receipt-layer0-component PRIVATE -ffp-contract=off)
target_include_directories(receipt-layer0-component PRIVATE "${CRISPEMBED_SOURCE}/src")
target_link_libraries(receipt-layer0-component PRIVATE crispembed-qwen35-delta crispembed-core)

add_executable(receipt-projection-component projection.cpp)
target_include_directories(receipt-projection-component PRIVATE "${CRISPEMBED_SOURCE}/src")
target_link_libraries(receipt-projection-component PRIVATE ggml ggml-base ggml-cpu)

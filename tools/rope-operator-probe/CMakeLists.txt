cmake_minimum_required(VERSION 3.16)
project(lighton4b_rope_operator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(ROPE_NATIVE_CAPTURE "Direct test-only native post-RoPE observation" OFF)
if(ROPE_NATIVE_CAPTURE)
 if(NOT DEFINED CRISPEMBED_SOURCE OR NOT EXISTS "${OBSERVED_ATTENTION}")
  message(FATAL_ERROR "Exact native source and generated observed attention required")
 endif()
 add_subdirectory("${CRISPEMBED_SOURCE}" product EXCLUDE_FROM_ALL)
 add_executable(rope-native-capture "${CRISPEMBED_SOURCE}/tools/attention252-probe/native.cpp" "${OBSERVED_ATTENTION}")
 target_include_directories(rope-native-capture PRIVATE "${CRISPEMBED_SOURCE}/src")
 target_link_libraries(rope-native-capture PRIVATE crispembed-qwen35-delta crispembed-core)
 target_compile_options(rope-native-capture PRIVATE -ffp-contract=off)
else()
 if(NOT DEFINED GGML_SOURCE OR NOT EXISTS "${GGML_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Exact pinned backend required")
 endif()
 add_subdirectory("${GGML_SOURCE}" ggml)
 add_executable(rope-operator rope.cpp)
 target_link_libraries(rope-operator PRIVATE ggml ggml-cpu)
endif()

cmake_minimum_required(VERSION 3.16)
project(lighton4b_physical_rope_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED CRISPEMBED_SOURCE)
 message(FATAL_ERROR "Exact candidate source required")
endif()
add_subdirectory("${CRISPEMBED_SOURCE}" product EXCLUDE_FROM_ALL)
add_executable(physical-rope-source-component component.cpp)
target_compile_options(physical-rope-source-component PRIVATE -ffp-contract=off)
target_include_directories(physical-rope-source-component PRIVATE "${CRISPEMBED_SOURCE}/src")
target_link_libraries(physical-rope-source-component PRIVATE crispembed-qwen35-delta crispembed-core)
find_package(Threads REQUIRED)
foreach(kind unsupported wasm-guard)
 add_executable(physical-rope-${kind} "${CRISPEMBED_SOURCE}/tests/test_lighton4b_physical_rope.cpp" "${CRISPEMBED_SOURCE}/src/lighton4b_physical_f16_attention.cpp")
 target_include_directories(physical-rope-${kind} PRIVATE "${CRISPEMBED_SOURCE}/src")
 target_link_libraries(physical-rope-${kind} PRIVATE ggml ggml-cpu Threads::Threads)
endforeach()
target_compile_definitions(physical-rope-wasm-guard PRIVATE __EMSCRIPTEN__=1 CRISPEMBED_LIGHTON4B_F16_SGEMM_ABI=1)

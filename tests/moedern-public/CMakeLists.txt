cmake_minimum_required(VERSION 3.20)
project(crispembed_moedern_public_build LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
# Model-free compile of the public draft as its own shared library. The root
# library is not changed here; observer macros are never defined.
add_library(moedern-public SHARED "${CRISPEMBED_ROOT}/src/core/gguf_loader.cpp"
  "${CRISPEMBED_ROOT}/src/moedern_text_engine.cpp" "${CRISPEMBED_ROOT}/src/moedern_text_frontend.cpp"
  "${CRISPEMBED_ROOT}/src/moedern_text.cpp" "${CRISPEMBED_ROOT}/src/moedern_text_api.cpp")
target_include_directories(moedern-public PUBLIC "${CRISPEMBED_ROOT}/src")
target_compile_definitions(moedern-public PRIVATE CRISPEMBED_BUILD)
target_link_libraries(moedern-public PRIVATE ggml ggml-cpu)
set_target_properties(moedern-public PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(moedern-public PRIVATE -ffp-contract=off)
endif()
add_executable(moedern-public-controls controls.c)
target_link_libraries(moedern-public-controls PRIVATE moedern-public)

cmake_minimum_required(VERSION 3.20)
project(crispembed_josev_quality_audit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_DIRECTORY "${REVIEWED_RUNTIME}/src")
    message(FATAL_ERROR "Explicit reviewed F32/Q4 runtime checkout required; no arbitrary main fallback")
endif()
set(CRISPEMBED_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(CRISPEMBED_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(CRISPEMBED_SYSTEMONE_AUDIT ON CACHE BOOL "" FORCE)
add_subdirectory("${REVIEWED_RUNTIME}" runtime)
if(NOT TARGET systemone-audit-observed OR NOT TARGET systemone-audit-plain)
    message(FATAL_ERROR "Reviewed root-profile observer libraries required")
endif()
foreach(variant plain observed)
    add_executable(josev-quality-${variant}-probe probe.cpp)
    target_include_directories(josev-quality-${variant}-probe PRIVATE "${REVIEWED_RUNTIME}/tests")
    target_link_libraries(josev-quality-${variant}-probe PRIVATE systemone-audit-${variant})
    crispembed_apply_arch_flags(josev-quality-${variant}-probe)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(josev-quality-${variant}-probe PRIVATE -ffp-contract=off)
    endif()
endforeach()
target_compile_definitions(josev-quality-observed-probe PRIVATE CRISPEMBED_JOSEV_COMPOSED_TESTING)
add_executable(josev-quality-diff diff.cpp)
target_include_directories(josev-quality-diff PRIVATE "${REVIEWED_RUNTIME}/src")

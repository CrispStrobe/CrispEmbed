cmake_minimum_required(VERSION 3.20)
project(crispembed_moedern_text LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_CPU OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
foreach(mode plain tested)
  add_library(moedern-text-${mode} STATIC "${CRISPEMBED_ROOT}/src/moedern_text_frontend.cpp")
  target_include_directories(moedern-text-${mode} PUBLIC "${CRISPEMBED_ROOT}/src")
  target_link_libraries(moedern-text-${mode} PUBLIC ggml-base)
  add_executable(moedern-text-${mode}-probe probe.cpp)
  target_link_libraries(moedern-text-${mode}-probe PRIVATE moedern-text-${mode})
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(moedern-text-${mode} PRIVATE -ffp-contract=off)
  endif()
endforeach()
target_compile_definitions(moedern-text-tested PRIVATE CRISPEMBED_MOEDERN_TEXT_TESTING)
target_compile_definitions(moedern-text-tested-probe PRIVATE CRISPEMBED_MOEDERN_TEXT_TESTING)

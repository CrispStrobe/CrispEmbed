cmake_minimum_required(VERSION 3.20)
project(moedern_first_row_attribution LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ROOT_BUILD)
  message(FATAL_ERROR "ROOT_BUILD must name the actual reviewed root-profile build")
endif()
find_library(GGML_BASE NAMES ggml-base PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_CPU NAMES ggml-cpu PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML NAMES ggml PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_package(OpenMP REQUIRED)
# Mirror exactly root crispembed_apply_arch_flags with CRISPEMBED_NATIVE=OFF.
file(STRINGS "${ROOT_BUILD}/CMakeCache.txt" ROOT_CACHE)
set(ARCH_FLAGS "")
foreach(option GGML_AVX512 GGML_AVX2 GGML_FMA GGML_F16C)
  if("${ROOT_CACHE}" MATCHES "${option}:BOOL=ON")
    if(option STREQUAL "GGML_AVX512")
      list(APPEND ARCH_FLAGS -mavx512f -mavx512cd -mavx512vl -mavx512dq -mavx512bw)
    elseif(option STREQUAL "GGML_AVX2")
      list(APPEND ARCH_FLAGS -mavx2)
    elseif(option STREQUAL "GGML_FMA")
      list(APPEND ARCH_FLAGS -mfma)
    elseif(option STREQUAL "GGML_F16C")
      list(APPEND ARCH_FLAGS -mf16c)
    endif()
  endif()
endforeach()
foreach(mode plain observed)
  add_library(first-row-${mode} STATIC "${ROOT}/src/moedern_text_engine.cpp"
    "${ROOT}/src/moedern_family_engine.cpp" "${ROOT}/src/moedern_text_frontend.cpp"
    "${ROOT}/src/moedern_text.cpp" "${ROOT}/src/core/gguf_loader.cpp")
  target_include_directories(first-row-${mode} PUBLIC "${ROOT}/src" "${ROOT}/ggml/include")
  target_compile_definitions(first-row-${mode} PRIVATE CRISPEMBED_MOEDERN_FAMILY_ENGINE)
  target_compile_options(first-row-${mode} PRIVATE -ffp-contract=off ${ARCH_FLAGS})
  target_link_libraries(first-row-${mode} PUBLIC "-Wl,--start-group" "${GGML}" "${GGML_CPU}" "${GGML_BASE}" "-Wl,--end-group" OpenMP::OpenMP_CXX pthread dl)
  add_executable(first-row-${mode}-probe probe.cpp)
  target_link_libraries(first-row-${mode}-probe PRIVATE first-row-${mode})
endforeach()
target_compile_definitions(first-row-observed PRIVATE CRISPEMBED_MOEDERN_COMPOSED_TESTING)
target_compile_definitions(first-row-observed-probe PRIVATE CRISPEMBED_MOEDERN_COMPOSED_TESTING)
add_executable(first-row-diff "${ROOT}/tests/moedern-text/diff.cpp")
target_include_directories(first-row-diff PRIVATE "${ROOT}/src")

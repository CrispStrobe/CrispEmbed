cmake_minimum_required(VERSION 3.20)
project(wasm_layer3_ffn_matmul LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Same static a06 GGML as the frozen controls, except GGML_LLAMAFILE follows the
# production libraries: ON for native libcrispembed, OFF for the embed WASM build.
# The caller admits the kernel files' arithmetic flags against the production
# full inventories before anything runs.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
if(EMSCRIPTEN)
 set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
else()
 set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/ggml" ggml-build)
add_executable(ffn-matmul-probe probe.cpp)
target_include_directories(ffn-matmul-probe PRIVATE "${ROOT}/ggml/src")
target_link_libraries(ffn-matmul-probe PRIVATE ggml ggml-cpu)
target_compile_options(ffn-matmul-probe PRIVATE -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(ffn-matmul-probe PRIVATE -pthread)
 target_link_options(ffn-matmul-probe PRIVATE -pthread -sENVIRONMENT=node -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=134217728
  -sEXIT_RUNTIME=1 -sFORCE_FILESYSTEM=1 -lnodefs.js)
 set_target_properties(ffn-matmul-probe PROPERTIES SUFFIX ".js")
endif()

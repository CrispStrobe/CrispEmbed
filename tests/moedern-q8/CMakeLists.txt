cmake_minimum_required(VERSION 3.20)
project(crispembed_moedern_q8 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
# Match the actual root profile; the historical standalone OFF path stays intact.
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
add_library(moedern-q8 SHARED "${CRISPEMBED_ROOT}/src/core/gguf_loader.cpp"
  "${CRISPEMBED_ROOT}/src/moedern_text_engine.cpp" "${CRISPEMBED_ROOT}/src/moedern_family_engine.cpp"
  "${CRISPEMBED_ROOT}/src/moedern_text_frontend.cpp" "${CRISPEMBED_ROOT}/src/moedern_text.cpp"
  "${CRISPEMBED_ROOT}/src/moedern_text_api.cpp")
target_include_directories(moedern-q8 PUBLIC "${CRISPEMBED_ROOT}/src")
target_compile_definitions(moedern-q8 PRIVATE CRISPEMBED_BUILD CRISPEMBED_MOEDERN_FAMILY_ENGINE)
target_link_libraries(moedern-q8 PRIVATE ggml ggml-cpu)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(moedern-q8 PRIVATE -ffp-contract=off)
endif()
add_executable(moedern-q8-controls controls.cpp)
target_link_libraries(moedern-q8-controls PRIVATE moedern-q8 ggml ggml-cpu)
add_executable(moedern-q8-quantize "${CRISPEMBED_ROOT}/tools/quantize.cpp")
target_link_libraries(moedern-q8-quantize PRIVATE ggml ggml-cpu)
add_executable(moedern-q8-consumer "${CRISPEMBED_ROOT}/tests/moedern-public/consumer.c")
target_link_libraries(moedern-q8-consumer PRIVATE moedern-q8)

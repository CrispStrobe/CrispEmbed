cmake_minimum_required(VERSION 3.20)
project(crispembed_moedern_native_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Standalone, test-only build. No root CMake or public runtime registration.
if(NOT DEFINED GGML_SOURCE)
  set(GGML_SOURCE "${CRISPEMBED_ROOT}/ggml")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml-build)
add_library(moedern-native STATIC
  "${CRISPEMBED_ROOT}/src/moedern_colbert.cpp"
  "${CRISPEMBED_ROOT}/src/core/gguf_loader.cpp")
target_include_directories(moedern-native PUBLIC "${CRISPEMBED_ROOT}/src")
target_link_libraries(moedern-native PUBLIC ggml ggml-cpu)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(moedern-native PRIVATE -ffp-contract=off)
endif()
add_executable(moedern-native-probe probe.cpp)
target_link_libraries(moedern-native-probe PRIVATE moedern-native)
add_executable(moedern-native-controls controls.cpp)
target_link_libraries(moedern-native-controls PRIVATE moedern-native)

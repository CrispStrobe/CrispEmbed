cmake_minimum_required(VERSION 3.20)
project(crispembed_josev_head_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT DEFINED GGML_SOURCE)
  set(GGML_SOURCE "${CRISPEMBED_ROOT}/ggml")
endif()
option(GGML_CUDA "Build explicit CUDA head comparison backend" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml-build)
add_library(josev-head-common STATIC "${CRISPEMBED_ROOT}/src/core/gguf_loader.cpp")
target_include_directories(josev-head-common PUBLIC "${CRISPEMBED_ROOT}/src")
target_link_libraries(josev-head-common PUBLIC ggml ggml-cpu)
if(GGML_CUDA)
  target_link_libraries(josev-head-common PUBLIC ggml-cuda)
endif()
# Ordinary archive must contain no observer symbols or reference-stage labels.
add_library(josev-head-plain STATIC "${CRISPEMBED_ROOT}/src/josev2b_head.cpp")
target_link_libraries(josev-head-plain PUBLIC josev-head-common)
add_library(josev-head-observed STATIC "${CRISPEMBED_ROOT}/src/josev2b_head.cpp")
target_link_libraries(josev-head-observed PUBLIC josev-head-common)
target_compile_definitions(josev-head-observed PRIVATE CRISPEMBED_JOSEV_HEAD_TESTING)
add_executable(test-josev2b-head "${CRISPEMBED_ROOT}/tests/test_josev2b_head_probe.cpp")
target_include_directories(test-josev2b-head PRIVATE "${CRISPEMBED_ROOT}/tests")
target_link_libraries(test-josev2b-head PRIVATE josev-head-observed)
add_executable(josev-head-diff diff.cpp)
target_include_directories(josev-head-diff PRIVATE "${CRISPEMBED_ROOT}/src")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target josev-head-common josev-head-plain josev-head-observed test-josev2b-head josev-head-diff)
    target_compile_options(${target} PRIVATE -ffp-contract=off)
  endforeach()
endif()

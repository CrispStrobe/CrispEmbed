cmake_minimum_required(VERSION 3.20)
project(portfolio_q8_profiles LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option GGML_NATIVE GGML_BLAS GGML_CUDA GGML_METAL GGML_VULKAN GGML_OPENCL GGML_BACKEND_DL GGML_BUILD_TESTS GGML_BUILD_EXAMPLES)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/ggml" ggml)
add_library(profile-common STATIC "${ROOT}/src/core/gguf_loader.cpp")
target_include_directories(profile-common PUBLIC "${ROOT}/src")
target_link_libraries(profile-common PUBLIC ggml ggml-cpu)
add_executable(q8-josev-profile josev.cpp)
target_link_libraries(q8-josev-profile PRIVATE profile-common)
add_executable(q8-ultrabert-profile ultrabert.cpp)
target_link_libraries(q8-ultrabert-profile PRIVATE profile-common)
add_executable(q8-quantize "${ROOT}/tools/quantize.cpp")
target_link_libraries(q8-quantize PRIVATE profile-common)
add_executable(q8-quantizer-fixture "${ROOT}/tests/gguf-file-type/fixture.cpp")
target_link_libraries(q8-quantizer-fixture PRIVATE profile-common)

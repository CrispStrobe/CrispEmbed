cmake_minimum_required(VERSION 3.20)
project(crispembed_gguf_file_type LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option GGML_NATIVE GGML_BACKEND_DL GGML_BUILD_TESTS GGML_BUILD_EXAMPLES GGML_CUDA GGML_METAL GGML_VULKAN GGML_OPENCL GGML_BLAS)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
add_executable(gguf-file-type-quantize "${CRISPEMBED_ROOT}/tools/quantize.cpp")
target_link_libraries(gguf-file-type-quantize PRIVATE ggml ggml-cpu)
add_executable(gguf-file-type-fixture fixture.cpp)
target_include_directories(gguf-file-type-fixture PRIVATE "${CRISPEMBED_ROOT}/src")
target_link_libraries(gguf-file-type-fixture PRIVATE ggml ggml-cpu)

cmake_minimum_required(VERSION 3.20)
project(moedern_scifact_replay LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ROOT_BUILD)
  message(FATAL_ERROR "ROOT_BUILD must identify the reviewed root-profile build")
endif()
find_library(GGML_BASE NAMES ggml-base PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_CPU NAMES ggml-cpu PATHS "${ROOT_BUILD}/ggml/src/ggml-cpu" NO_DEFAULT_PATH REQUIRED)
find_library(GGML NAMES ggml PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(CRISP NAMES crispembed PATHS "${ROOT_BUILD}" NO_DEFAULT_PATH REQUIRED)
find_package(OpenMP REQUIRED)
add_library(replay-family STATIC "${ROOT}/src/moedern_family_engine.cpp" "${ROOT}/src/core/gguf_loader.cpp")
target_include_directories(replay-family PUBLIC "${ROOT}/src" "${ROOT}/ggml/include")
target_link_libraries(replay-family PUBLIC "-Wl,--start-group" "${GGML}" "${GGML_CPU}" "${GGML_BASE}" "-Wl,--end-group" OpenMP::OpenMP_CXX pthread dl)
target_compile_options(replay-family PRIVATE -ffp-contract=off)
add_executable(replay-family-control "${ROOT}/tests/moedern-q4k/family_control.cpp")
target_link_libraries(replay-family-control PRIVATE replay-family)
add_executable(replay-controls "${ROOT}/tests/moedern-q4k/controls.cpp")
target_link_libraries(replay-controls PRIVATE replay-family)
add_executable(replay-tokens tokenize.cpp "${ROOT}/src/moedern_text_frontend.cpp")
target_include_directories(replay-tokens PRIVATE "${ROOT}/src" "${ROOT}/ggml/include")
target_compile_options(replay-tokens PRIVATE -ffp-contract=off)
target_link_libraries(replay-tokens PRIVATE "${GGML_BASE}")
add_executable(replay-consumer "${ROOT}/tests/moedern-public/consumer.c")
target_include_directories(replay-consumer PRIVATE "${ROOT}/src")
target_link_libraries(replay-consumer PRIVATE "${CRISP}")
add_executable(replay-recovery "${ROOT}/tests/moedern-q4k/recovery.cpp")
target_include_directories(replay-recovery PRIVATE "${ROOT}/src")
target_link_libraries(replay-recovery PRIVATE "${CRISP}")
add_executable(replay-diff diff.cpp)
target_include_directories(replay-diff PRIVATE "${ROOT}/src")
add_executable(replay-authored-diff "${ROOT}/tests/moedern-text/diff.cpp")
target_include_directories(replay-authored-diff PRIVATE "${ROOT}/src")

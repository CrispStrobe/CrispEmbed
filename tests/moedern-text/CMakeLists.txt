cmake_minimum_required(VERSION 3.20)
project(crispembed_moedern_text_composed LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
add_library(moedern-text-common STATIC "${CRISPEMBED_ROOT}/src/core/gguf_loader.cpp")
target_include_directories(moedern-text-common PUBLIC "${CRISPEMBED_ROOT}/src")
target_link_libraries(moedern-text-common PUBLIC ggml ggml-cpu)
foreach(mode plain observed)
  add_library(moedern-text-${mode} STATIC "${CRISPEMBED_ROOT}/src/moedern_text_engine.cpp"
     "${CRISPEMBED_ROOT}/src/moedern_text_frontend.cpp" "${CRISPEMBED_ROOT}/src/moedern_text.cpp")
  target_link_libraries(moedern-text-${mode} PUBLIC moedern-text-common)
  add_executable(moedern-text-${mode}-probe probe.cpp)
  target_link_libraries(moedern-text-${mode}-probe PRIVATE moedern-text-${mode})
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(moedern-text-${mode} PRIVATE -ffp-contract=off)
  endif()
endforeach()
target_compile_definitions(moedern-text-observed PRIVATE CRISPEMBED_MOEDERN_COMPOSED_TESTING)
target_compile_definitions(moedern-text-observed-probe PRIVATE CRISPEMBED_MOEDERN_COMPOSED_TESTING)
add_executable(moedern-text-diff diff.cpp)
target_include_directories(moedern-text-diff PRIVATE "${CRISPEMBED_ROOT}/src")

add_executable(moedern-text-controls controls.cpp)
target_link_libraries(moedern-text-controls PRIVATE moedern-text-plain)

cmake_minimum_required(VERSION 3.20)
project(wasm_native_order_norm_controls LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/ggml" ggml-build)
add_executable(norm-candidate-controls controls.cpp "${ROOT}/src/core/wasm_native_order_norm.cpp")
target_include_directories(norm-candidate-controls PRIVATE "${ROOT}/src")
target_link_libraries(norm-candidate-controls PRIVATE ggml ggml-cpu)
target_compile_options(norm-candidate-controls PRIVATE -ffp-contract=off)
if(EMSCRIPTEN)
 target_compile_options(norm-candidate-controls PRIVATE -pthread)
 target_link_options(norm-candidate-controls PRIVATE -pthread -sMODULARIZE=1 -sEXPORT_NAME=NativeOrderNormControls
  -sENVIRONMENT=web,worker -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=67108864 -sPTHREAD_POOL_SIZE=2
  "-sEXPORTED_FUNCTIONS=['_norm_candidate_controls','_norm_candidate_replay','_malloc','_free']"
  "-sEXPORTED_RUNTIME_METHODS=['ccall','HEAPF32','HEAPU8']")
 set_target_properties(norm-candidate-controls PROPERTIES SUFFIX ".js")
endif()
add_executable(norm-candidate-ref "${ROOT}/tests/moedern-native-v2/diff.cpp")
target_include_directories(norm-candidate-ref PRIVATE "${ROOT}/src")
target_compile_options(norm-candidate-ref PRIVATE -ffp-contract=off)

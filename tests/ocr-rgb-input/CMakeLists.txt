cmake_minimum_required(VERSION 3.20)
project(crispembed_rgb_input_audit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "Declared portable Linux x64 production baseline required")
endif()
if(NOT EXISTS "${PRODUCTION_PREFIX}" OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "Authenticated production prefix and Release profile required")
endif()
# Same fixed x86 flags selected by crispembed_apply_arch_flags with native OFF
# and the pinned ggml AVX2/FMA/F16C baseline. No STBI overrides or fast-math.
add_library(production-stb STATIC "${PRODUCTION_PREFIX}")
target_include_directories(production-stb PRIVATE "${ROOT}/src")
target_compile_options(production-stb PRIVATE -mavx2 -mfma -mf16c)
add_executable(ocr-rgb-input probe.cpp)
target_include_directories(ocr-rgb-input PRIVATE "${ROOT}/src" "${ROOT}/ggml/examples")
target_link_libraries(ocr-rgb-input PRIVATE production-stb)

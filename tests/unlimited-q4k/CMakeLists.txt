cmake_minimum_required(VERSION 3.18)
project(crispembed_unlimited_q4_audit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CRISPEMBED_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${CRISPEMBED_ROOT}" root)
add_executable(unlimited-q4-projection projection.cpp)
target_include_directories(unlimited-q4-projection PRIVATE "${CRISPEMBED_ROOT}/src")
target_link_libraries(unlimited-q4-projection PRIVATE crispembed-core ggml ggml-cpu ggml-base)
add_executable(unlimited-q4-diff diff.cpp)
target_include_directories(unlimited-q4-diff PRIVATE "${CRISPEMBED_ROOT}/src")
foreach(target unlimited-q4-projection unlimited-q4-diff)
    crispembed_apply_arch_flags(${target})
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -ffp-contract=off)
    endif()
endforeach()

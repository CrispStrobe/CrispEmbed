cmake_minimum_required(VERSION 3.20)
project(moedern_real_router_component LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT ROOT_BUILD)
  message(FATAL_ERROR "ROOT_BUILD must name the actual reviewed root-profile build")
endif()
find_library(GGML_BASE NAMES ggml-base PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_CPU NAMES ggml-cpu PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML NAMES ggml PATHS "${ROOT_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_package(OpenMP REQUIRED)
# Mirror exactly root crispembed_apply_arch_flags with CRISPEMBED_NATIVE=OFF.
file(STRINGS "${ROOT_BUILD}/CMakeCache.txt" ROOT_CACHE)
set(ARCH_FLAGS "")
foreach(option GGML_AVX512 GGML_AVX2 GGML_FMA GGML_F16C)
  if("${ROOT_CACHE}" MATCHES "${option}:BOOL=ON")
    if(option STREQUAL "GGML_AVX512")
      list(APPEND ARCH_FLAGS -mavx512f -mavx512cd -mavx512vl -mavx512dq -mavx512bw)
    elseif(option STREQUAL "GGML_AVX2")
      list(APPEND ARCH_FLAGS -mavx2)
    elseif(option STREQUAL "GGML_FMA")
      list(APPEND ARCH_FLAGS -mfma)
    elseif(option STREQUAL "GGML_F16C")
      list(APPEND ARCH_FLAGS -mf16c)
    endif()
  endif()
endforeach()
if(NOT FRAGMENT_DIR)
  message(FATAL_ERROR "FRAGMENT_DIR must hold the exact authenticated native source fragment")
endif()
add_executable(router-real-probe probe.cpp "${ROOT}/src/core/gguf_loader.cpp")
target_include_directories(router-real-probe PRIVATE "${ROOT}/src" "${ROOT}/ggml/include" "${FRAGMENT_DIR}")
target_compile_options(router-real-probe PRIVATE -ffp-contract=off ${ARCH_FLAGS})
target_link_libraries(router-real-probe PRIVATE "-Wl,--start-group" "${GGML}" "${GGML_CPU}" "${GGML_BASE}" "-Wl,--end-group" OpenMP::OpenMP_CXX pthread dl)
add_executable(router-real-ref "${ROOT}/tests/moedern-native-v2/diff.cpp")
target_include_directories(router-real-ref PRIVATE "${ROOT}/src")
target_compile_options(router-real-ref PRIVATE -ffp-contract=off)

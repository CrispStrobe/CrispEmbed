cmake_minimum_required(VERSION 3.20)
project(moedern_router_topk_v1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(CRISPEMBED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory("${CRISPEMBED_ROOT}/ggml" ggml-build)
add_executable(router-topk-operator probe.cpp)
target_include_directories(router-topk-operator PRIVATE "${CRISPEMBED_ROOT}/src")
target_link_libraries(router-topk-operator PRIVATE ggml ggml-cpu)
add_executable(router-topk-ref "${CRISPEMBED_ROOT}/tests/moedern-native-v2/diff.cpp")
target_include_directories(router-topk-ref PRIVATE "${CRISPEMBED_ROOT}/src")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(router-topk-operator PRIVATE -ffp-contract=off)
  target_compile_options(router-topk-ref PRIVATE -ffp-contract=off)
endif()

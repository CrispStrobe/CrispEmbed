cmake_minimum_required(VERSION 3.20)
project(gemma_decision_q4_audit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CRISPEMBED_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/ggml" ggml-build)
if(NOT DEFINED ROOT_ARCH_HELPER OR NOT EXISTS "${ROOT_ARCH_HELPER}")
 message(FATAL_ERROR "Exact source-derived root CPU architecture helper required")
endif()
include("${ROOT_ARCH_HELPER}")
# The core target follows root's core precision flags; model/tokenizer mirror root's ISA helper.
add_library(gemma-audit-core STATIC "${ROOT}/src/core/gguf_loader.cpp" "${ROOT}/src/core/shortest_double.c")
target_include_directories(gemma-audit-core PUBLIC "${ROOT}/src" PRIVATE "${ROOT}/src/core/third_party")
target_link_libraries(gemma-audit-core PUBLIC ggml ggml-cpu)
add_library(gemma-audit-engine STATIC "${ROOT}/src/gemma_decision.cpp" "${ROOT}/src/tokenizer_bpe.cpp")
target_include_directories(gemma-audit-engine PUBLIC "${ROOT}/src")
target_link_libraries(gemma-audit-engine PUBLIC gemma-audit-core ggml ggml-cpu)
crispembed_apply_arch_flags(gemma-audit-engine)
add_executable(gemma-q4-probe probe.cpp)
target_link_libraries(gemma-q4-probe PRIVATE gemma-audit-engine)
add_executable(gemma-q4-schema "${ROOT}/tests/test_gemma_decision_schema.cpp")
target_link_libraries(gemma-q4-schema PRIVATE gemma-audit-core)
add_executable(gemma-q4-activation "${ROOT}/tests/test_gemma_decision_activation.cpp")
target_link_libraries(gemma-q4-activation PRIVATE gemma-audit-engine)
add_executable(crispembed-quantize "${ROOT}/tools/quantize.cpp")
target_link_libraries(crispembed-quantize PRIVATE ggml ggml-cpu)
